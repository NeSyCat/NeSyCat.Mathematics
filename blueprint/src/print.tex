% This file makes a printable version of the blueprint
% It should include all the \usepackage needed for the pdf version.
% The template version assume you want to use a modern TeX compiler
% such as xeLaTeX or luaLaTeX including support for unicode

% It also uses expl3 in order to support macros related to the dependency graph.
% It also includes standard AMS packages (and their improved version
% mathtools) as well as support for links with a sober decoration
% (no ugly rectangles around links).
% It is otherwise a very minimal preamble (you should probably at least
% add cleveref and tikz-cd).

\documentclass[a4paper]{report}

% C2-E3 overfull-lines sweep: a global, print-only tolerance knob.
% \emergencystretch gives the line breaker extra last-resort stretch
% before it gives up and overfills a line -- this absorbs small
% overfull \hbox warnings (a few points) from ordinary text paragraphs
% without touching any content; \tolerance is loosened one notch from
% LaTeX's default (200) so a borderline line is allowed to break rather
% than overfill. Neither setting affects the web build (web.tex does
% not \input this file). Genuinely wide material (long unbreakable
% identifier chains, display formulas wider than the text width, the
% linear-logic dictionary table) is fixed at the call site instead --
% this knob only mops up ordinary-paragraph residue.
\setlength{\emergencystretch}{3em}
\tolerance=500

\usepackage{geometry}

\usepackage{expl3}

\usepackage{amssymb, amsthm, mathtools}
% C2-E8, Law 2 (visible attribution): natbib for \citep/\citet
% author-year citations, matching the user's own papers' citation
% style (new.tex, nesy2026-paper.tex both auto-load natbib with no
% explicit options -- its own default is already round-paren,
% author-year). No \bibliographystyle/\bibliography{file} pair is
% used: the bibliography is a hand-written thebibliography (see
% content.tex's Bibliography section) with natbib-format \bibitem[...]
% optional arguments, so no bibtex/biber run is required; \bibitem
% itself writes \bibcite to the .aux on every compile, and latexmkrc's
% automatic rerun-until-stable resolves it exactly like \ref/\label.
\usepackage{natbib}
\usepackage[unicode,colorlinks=true,linkcolor=blue,urlcolor=magenta, citecolor=blue]{hyperref}

\usepackage[warnings-off={mathtools-colon,mathtools-overbracket}]{unicode-math}

\input{macros/common}
\input{macros/print}

\title{NeSyCat: Neuro Symbolic Categories}
\author{Daniel Romero Schellhorn}
